# --- PAL-M video standard -------------------------------------------------
# The captured Windows session leaves the decoder in NTSC-M and never touches
# 0x0403 at all, so selecting a PAL format alone gives line lock with the wrong
# colour. cx25840-core.c set_v4lstd() does FOUR things for V4L2_STD_PAL_M, and
# all four are needed. Measured starting values on this board:
#     0x0400 = 0x21   (low nibble 1 = NTSC-M)
#     0x0403 = 0x00   (pal_m clear)
#     0x047b = 0x66   (LCOMB set in bits 2:1)
#
# 1. force NTSC-M first, per step 9 of datasheet section 3.16
CMD 08 44 03 04 00 21
# 2. turn off LCOMB: 0x47b &= ~6  ->  0x66 & ~0x06 = 0x60
CMD 08 44 03 04 7b 60
# 3. select PAL-M: 0x400 low nibble -> 5
CMD 08 44 03 04 00 25
# 4. set the pal_m chroma bit: 0x403 bits 1:0 -> 1
CMD 08 44 03 04 03 01
